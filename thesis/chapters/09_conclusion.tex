\chapter{Συμπέρασμα}

\section{Περίληψη της εργασίας}

Η παρούσα εργασία διερεύνησε την ταξινόμηση με βάση τη φωνή της νόσου του Πάρκινσον (PD) σε σύγκριση με υγιείς μάρτυρες (HC) χρησιμοποιώντας κλασικές μεθόδους μηχανικής μάθησης. Η εργασία ασχολήθηκε με βασικές μεθοδολογικές προκλήσεις στον τομέα, όπως διαρροή δεδομένων σε επίπεδο υποκειμένου, ανισορροπία κλάσεων και αναπαράσταση χαρακτηριστικών.

\subsection{Συνεισφορές}

\begin{enumerate}
    \item \textbf{Αυστηρό πλαίσιο αξιολόγησης:} Ομαδοποιημένη στρωματοποιημένη διασταυρούμενη επικύρωση με συστηματικό σχεδιασμό 2$\times$2 (χαρακτηριστικά $\times$ στάθμιση κατηγοριών) και διαφανή αναφορά σε όλες τις συνθήκες.
    \item \textbf{Έρευνα μηχανικής χαρακτηριστικών:} Επέκταση του συνόλου χαρακτηριστικών από 47 σε 78 ακουστικά χαρακτηριστικά, επιδεικνύοντας βελτίωση έως +23pp ROC-AUC και προσδιορίζοντας τα $F_0$, MFCCs και την αρμονικότητα ως τα πιο διακριτικά.
    \item \textbf{Ανάλυση στάθμισης κλάσης:} Αξιολόγηση \texttt{class\_weight=«balanced»} σε όλα τα μοντέλα, διαπιστώνοντας μέτρια και ασυνεπή αποτελέσματα σε μέτρια ανισορροπημένα δεδομένα.
    \item \textbf{Αναπαραγώγιμη ροή εργασιών:} Εργαλεία CLI, σταθεροποιημένοι τυχαίοι σπόροι, τεκμηριωμένες παράμετροι και μια διεπαφή επίδειξης έρευνας χωρίς διαγνωστικά στοιχεία (Παράρτημα~\ref{ch:demo-app}).
\end{enumerate}

\section{Απαντηθέντα ερευνητικά ερωτήματα}

\subsection{RQ1: Πώς αποδίδουν τα κλασικά μοντέλα ML στην ταξινόμηση φωνής PD;}

Το κλασικό ML επιτυγχάνει \textbf{ROC-AUC έως $0{,}857 \pm 0{,}171$} (Random Forest, εκτεταμένα χαρακτηριστικά, SpontaneousDialogue) στο σύνολο δεδομένων Α με ομαδοποιημένη διασταυρούμενη επικύρωση. Ωστόσο, η απόδοση ποικίλλει σημαντικά:

\begin{itemize}
    \item ReadText: 0,590--0,834 ROC-AUC (βασικά χαρακτηριστικά, σε όλα τα μοντέλα)
    \item SpontaneousDialogue: 0,407--0,857 ROC-AUC (βασικά $\rightarrow$ εκτεταμένα, RF)
    \item Μεγάλη διακύμανση λόγω μικρού μεγέθους δείγματος ($n=37$ άτομα)
    \item Ορισμένες πτυχές SVM παρήγαγαν ROC-AUC $< 0{,}5$, υποδηλώνοντας αστάθεια
\end{itemize}

Αυτό αποδεικνύει τη \textbf{εφικτότητα} της ταξινόμησης PD με βάση τη φωνή, αλλά υπογραμμίζει την \textbf{πρόκληση} της σταθερής απόδοσης σε μικρά σύνολα δεδομένων.

\subsection{RQ2: Η επέκταση του συνόλου χαρακτηριστικών βελτιώνει την απόδοση της ταξινόμησης;}

\textbf{Εξαρτάται από την εργασία.} Επέκταση από 47 βασικά σε 78 εκτεταμένα χαρακτηριστικά:

\textbf{ReadText (Random Forest):}
\begin{itemize}
    \item Βασική γραμμή: $0{,}590 \pm 0{,}302$ ROC-AUC
    \item Επεκταμένα: $0{,}822 \pm 0{,}166$ ROC-AUC
    \item Βελτίωση: +23,3 ποσοστιαίες μονάδες (+39\% σχετική)
    \item Η διακύμανση μειώθηκε από 0,302 σε 0,166
\end{itemize}

\textbf{SpontaneousDialogue (Random Forest):}
\begin{itemize}
    \item Βασική γραμμή: $0{,}828 \pm 0{,}148$ ROC-AUC
    \item Επεκταμένη: $0{,}857 \pm 0{,}171$ ROC-AUC
    \item Βελτίωση: +2,9 ποσοστιαίες μονάδες (μέτρια)
\end{itemize}

Τα πρόσθετα χαρακτηριστικά που καταγράφουν τη φασματική μεταβλητότητα (MFCC std), τη χρονική δυναμική (delta-delta MFCC) και το φασματικό σχήμα συνέβαλαν σημαντικά στην απόδοση του ReadText. Τα αποτελέσματα ήταν λιγότερο έντονα για το SpontaneousDialogue, πιθανώς λόγω της υψηλότερης μεταβλητότητας της βασικής γραμμής στην αυθόρμητη ομιλία.

\subsection{RQ3: Βελτιώνει η στάθμιση των κλάσεων την απόδοση σε μη ισορροπημένα σύνολα δεδομένων;}

\textbf{Παρατηρήθηκαν ελάχιστα αποτελέσματα.} Στο σύνολο δεδομένων Α (αναλογία HC:PD 1,31:1 για το ReadText, 1,40:1 για το SpontaneousDialogue), η στάθμιση των κλάσεων έδειξε ασυνεπή αποτελέσματα:

\textbf{ReadText (Random Forest, εκτεταμένα χαρακτηριστικά):}
\begin{itemize}
    \item Χωρίς στάθμιση: $0{,}822 \pm 0{,}166$ ROC-AUC
    \item Με στάθμιση: $0{,}805 \pm 0{,}182$ ROC-AUC
    \item Αλλαγή: -1,7pp (ελαφρά μείωση)
\end{itemize}

\textbf{SpontaneousDialogue (Random Forest, βασικά χαρακτηριστικά):}
\begin{itemize}
    \item Χωρίς στάθμιση: $0{,}828 \pm 0{,}148$ ROC-AUC
    \item Με στάθμιση: $0{,}827 \pm 0{,}133$ ROC-AUC
    \item Αλλαγή: -0,1pp (αμελητέα)
\end{itemize}

Η μέτρια ανισορροπία (1,3:1) μπορεί να μην είναι αρκετά σοβαρή ώστε να ωφεληθεί από τη στάθμιση. Επιπλέον, με μόνο 37 άτομα, η στάθμιση των κατηγοριών μπορεί να εισαγάγει επιπλέον διακύμανση. Τα αποτελέσματα ποίκιλαν μεταξύ των μοντέλων και των συνόλων χαρακτηριστικών, χωρίς να υπάρχει σταθερό μοτίβο βελτίωσης.

\subsection{RQ4: Πώς συγκρίνονται τα αποτελέσματα μεταξύ της ομαδοποιημένης και της τυπικής CV;}

\textbf{Σύνολο δεδομένων Α (ομαδοποιημένη CV, $n=37$ άτομα):}
\begin{itemize}
    \item Random Forest: 0,590--0,857 ROC-AUC
    \item Υψηλή διακύμανση (std 0,15--0,30)
    \item Γενίκευση σε επίπεδο υποκειμένου
\end{itemize}

\textbf{Σύνολο δεδομένων Β (Τυπική CV, $n=756$ δείγματα, άγνωστα αναγνωριστικά υποκειμένων):}
\begin{itemize}
    \item Random Forest: 0,940--0,949 ROC-AUC (βασική γραμμή/σταθμισμένη)
    \item Χαμηλή διακύμανση (std 0,012--0,013)
    \item Άγνωστη συσχέτιση εντός υποκειμένου
\end{itemize}

Το σύνολο δεδομένων Β παρουσιάζει σημαντικά υψηλότερη απόδοση (+8--12pp ROC-AUC) και χαμηλότερη διακύμανση, σε συμφωνία με την πιθανή αισιόδοξη μεροληψία από τη διαρροή υποκειμένων. \textbf{Η ομαδοποιημένη CV παρέχει πιο συντηρητικές αλλά πιο ρεαλιστικές εκτιμήσεις} της γενίκευσης εκτός υποκειμένου. Η σύγκριση περιπλέκεται από διαφορετικά μεγέθη δειγμάτων, χώρους χαρακτηριστικών και εργασίες ομιλίας, περιορίζοντας την άμεση ερμηνεία.

\subsection{RQ5: Οι εργασίες ομιλίας αποδίδουν διαφορετική απόδοση ταξινόμησης;}

\textbf{Τα βασικά χαρακτηριστικά παρουσιάζουν διαφορές στις εργασίες:}
\begin{itemize}
    \item ReadText: $0{,}590 \pm 0{,}302$ ROC-AUC (Random Forest, βασική γραμμή)
    \item SpontaneousDialogue: $0{,}828 \pm 0{,}148$ ROC-AUC (Random Forest, βασική γραμμή)
    \item Διαφορά: +23,8pp υπέρ της αυθόρμητης ομιλίας
\end{itemize}

\textbf{Τα εκτεταμένα χαρακτηριστικά μειώνουν το χάσμα:}
\begin{itemize}
    \item ReadText: $0{,}822 \pm 0{,}166$ (εκτεταμένο)
    \item SpontaneousDialogue: $0{,}857 \pm 0{,}171$ (επεκταμένο)
    \item Διαφορά: +3,5pp (συγκλίνουσα)
\end{itemize}

Το SpontaneousDialogue μπορεί να καταγράψει τα φυσιολογικά χαρακτηριστικά της ομιλίας PD (μονοτονική προσωδία, μειωμένη μεταβλητότητα) πιο αποτελεσματικά από την δομημένη ανάγνωση. Ωστόσο, το ReadText επωφελείται περισσότερο από την επέκταση χαρακτηριστικών, πιθανώς επειδή τα επεκταμένα χαρακτηριστικά καταγράφουν καλύτερα τη δομημένη δυναμική της ομιλίας.

\section{Επιπτώσεις}

\subsection{Για ερευνητές}

\begin{itemize}
    \item \textbf{Χρησιμοποιήστε ομαδοποιημένο CV} όταν υπάρχουν πολλαπλές ηχογραφήσεις ανά θέμα
    \item \textbf{Συμπεριλάβετε χαρακτηριστικά μεταβλητότητας} (std, delta-delta) στα σύνολα χαρακτηριστικών
    \item \textbf{Αναφέρετε όλες τις συνθήκες} αντί να επιλέγετε τα καλύτερα αποτελέσματα
    \item \textbf{Αναγνωρίστε τους περιορισμούς} με διαφάνεια
\end{itemize}

\subsection{Για τους επαγγελματίες}

\begin{itemize}
    \item Η φωνητική εξέταση για την PD είναι εφικτή, αλλά δεν έχει ακόμη κλινική αξιοπιστία
    \item Το Random Forest παρέχει μια ισχυρή βάση για παρόμοιες εργασίες
    \item Η ερμηνευσιμότητα των χαρακτηριστικών υποστηρίζει την κλινική κατανόηση
    \item Τα αποτελέσματα απαιτούν επικύρωση σε ανεξάρτητες ομάδες
\end{itemize}

\subsection{Για τους δημιουργούς συνόλων δεδομένων}

\begin{itemize}
    \item \textbf{Πάντα να περιλαμβάνετε αναγνωριστικά υποκειμένων} για να επιτρέψετε τη σωστή CV
    \item Τεκμηριώστε τις συνθήκες και τον εξοπλισμό καταγραφής
    \item Παρέχετε δημογραφικές πληροφορίες
    \item Εξετάστε το ενδεχόμενο διαχρονικών σχεδίων
\end{itemize}

\section{Μελλοντικές κατευθύνσεις}

Οι βραχυπρόθεσμες επεκτάσεις περιλαμβάνουν βελτιστοποίηση υπερπαραμέτρων με ένθετη διασταυρούμενη επικύρωση, επιλογή χαρακτηριστικών για τη μείωση της διαστατικότητας, σύνθεση πολλαπλών εργασιών (συνδυασμός ReadText και SpontaneousDialogue) και πρόσθετα ακουστικά χαρακτηριστικά (wavelets, TQWT). Η μεσοπρόθεσμη έρευνα θα πρέπει να δώσει προτεραιότητα στην εξωτερική επικύρωση σε ανεξάρτητες κοόρτες, στη διαχρονική παρακολούθηση της εξέλιξης της νόσου και στην ταξινόμηση της σοβαρότητας σε πολλές κατηγορίες. Μακροπρόθεσμα, η φωνητική παρακολούθηση της νόσου του Πάρκινσον θα μπορούσε να ενσωματωθεί σε εφαρμογές για smartphone μαζί με πολυτροπικούς βιοδείκτες (φωνή, βάδισμα, τρόμος) και να υποβληθεί σε προοπτική κλινική επικύρωση.

\section{Τελικές παρατηρήσεις}

Η παρούσα εργασία αποδεικνύει ότι \textbf{η ταξινόμηση της νόσου του Πάρκινσον με βάση τη φωνή είναι εφικτή} χρησιμοποιώντας κλασική μηχανική μάθηση με προσεκτικά σχεδιασμένα ακουστικά χαρακτηριστικά, επιτυγχάνοντας ROC-AUC έως 0,857 στο σύνολο δεδομένων Α με ομαδοποιημένη διασταυρούμενη επικύρωση. Δίνοντας προτεραιότητα στην \textbf{μεθοδολογική εγκυρότητα έναντι της βελτιστοποίησης της απόδοσης}---διαφανείς διαχωρισμοί σε επίπεδο υποκειμένου, σταθερές παράμετροι και πλήρης αναφορά διακύμανσης---η εργασία αυτή παρέχει μια επαναλήψιμη βάση πάνω στην οποία μελλοντικές μελέτες μπορούν να αντιμετωπίσουν τους προσδιορισμένους περιορισμούς του μικρού μεγέθους του δείγματος, της απουσίας εξωτερικής επικύρωσης και του δυαδικού πεδίου εφαρμογής.
